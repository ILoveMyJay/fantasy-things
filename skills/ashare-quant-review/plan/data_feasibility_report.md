### 2. 涨跌停判定逻辑（2026年新规适配）

```python
def get_limit_threshold(stock_code, stock_name):
    """根据股票代码和名称判定涨跌幅限制"""
    # 新股判定（上市前5日无涨跌幅限制）
    if stock_name.startswith('N') or stock_name.startswith('C'):
        return None  # 无限制
    
    # 板块判定
    if stock_code.startswith('688'):  # 科创板
        return 0.20
    elif stock_code.startswith('300'):  # 创业板
        return 0.20
    elif stock_code.startswith('43') or stock_code.startswith('83'):  # 北交所
        return 0.30
    else:  # 主板
        # 2026年7月新规：主板ST涨跌幅从5%升至10%
        return 0.10

def is_limit_up(high, prev_close, threshold):
    """判定是否涨停（容差0.02元避免浮点误差）"""
    if threshold is None:
        return False
    limit_price = round(prev_close * (1 + threshold), 2)
    return abs(high - limit_price) <= 0.02

def is_limit_down(low, prev_close, threshold):
    """判定是否跌停"""
    if threshold is None:
        return False
    limit_price = round(prev_close * (1 - threshold), 2)
    return abs(low - limit_price) <= 0.02
```
